cmake_minimum_required(VERSION 3.20)
project(huoguo_native_prototypes LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MOONLIGHT_SOURCE_DIR)
  message(FATAL_ERROR "Use build_native.py to supply the pinned external Moonlight checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MOONLIGHT_SOURCE_DIR}" moonlight-core)
add_executable(touch_probe native/touch_probe.cpp)
target_compile_options(touch_probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(fec_probe native/fec_probe.cpp)
target_include_directories(fec_probe SYSTEM PRIVATE
  "${MOONLIGHT_SOURCE_DIR}/nanors" "${MOONLIGHT_SOURCE_DIR}/nanors/deps/obl")
target_link_libraries(fec_probe PRIVATE moonlight-common-c)
target_compile_options(fec_probe PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME android_native_touch COMMAND touch_probe)
add_test(NAME moonlight_reed_solomon COMMAND fec_probe)
