cmake_minimum_required(VERSION 3.20)
project(huoguo_android_udp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MOONLIGHT_SOURCE_DIR)
  message(FATAL_ERROR "Pass the verified external pinned Moonlight checkout")
endif()
add_library(pinned_nanors STATIC
  "${MOONLIGHT_SOURCE_DIR}/nanors/rs.c"
  "${MOONLIGHT_SOURCE_DIR}/nanors/deps/obl/oblas_common.c"
  "${MOONLIGHT_SOURCE_DIR}/nanors/deps/obl/oblas_lite.c")
set_target_properties(pinned_nanors PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(pinned_nanors SYSTEM PUBLIC
  "${MOONLIGHT_SOURCE_DIR}/nanors" "${MOONLIGHT_SOURCE_DIR}/nanors/deps/obl")
find_package(Threads REQUIRED)
if(ANDROID)
  add_library(huoguo_udp_fec SHARED native_udp_fec.cpp)
  target_link_libraries(huoguo_udp_fec PRIVATE pinned_nanors Threads::Threads)
  target_compile_options(huoguo_udp_fec PRIVATE -Wall -Wextra -Werror)
else()
  add_executable(h264_udp_packetizer h264_udp_packetizer.cpp)
  target_link_libraries(h264_udp_packetizer PRIVATE pinned_nanors Threads::Threads)
  target_compile_options(h264_udp_packetizer PRIVATE -Wall -Wextra -Werror)
  add_executable(phone_receiver_probe phone_receiver_probe.cpp)
  target_link_libraries(phone_receiver_probe PRIVATE pinned_nanors Threads::Threads)
  target_compile_options(phone_receiver_probe PRIVATE -Wall -Wextra -Werror)
  enable_testing()
  add_test(NAME phone_clock_fec_replay COMMAND phone_receiver_probe --self-test)
endif()
