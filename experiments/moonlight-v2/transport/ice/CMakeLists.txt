cmake_minimum_required(VERSION 3.18)
project(huoguo_ice_udp_backend LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT LIBJUICE_SOURCE_DIR)
  message(FATAL_ERROR "An exact pinned, patched external LIBJUICE_SOURCE_DIR is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NO_SERVER ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(USE_NETTLE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBJUICE_SOURCE_DIR}" libjuice EXCLUDE_FROM_ALL)
set_target_properties(juice-static PROPERTIES C_VISIBILITY_PRESET hidden)
add_library(huoguo_ice_udp SHARED ice_udp_backend.c)
set_target_properties(huoguo_ice_udp PROPERTIES C_VISIBILITY_PRESET hidden)
target_compile_definitions(huoguo_ice_udp PRIVATE JUICE_STATIC)
target_compile_options(huoguo_ice_udp PRIVATE -Wall -Wextra -Werror)
target_include_directories(huoguo_ice_udp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(huoguo_ice_udp PRIVATE juice-static)
if(NOT ANDROID)
  add_executable(ice_local_probe ice_local_probe.c)
  target_compile_definitions(ice_local_probe PRIVATE JUICE_STATIC)
  target_compile_options(ice_local_probe PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ice_local_probe PRIVATE huoguo_ice_udp juice-static)
endif()
